11. Установка Node.js через nvm (без sudo)

Системный Node.js в РЕД ОС ставится через dnf — а это требует sudo. Используем nvm (Node Version Manager) — устанавливается в ~/.nvm, позволяет держать несколько версий Node одновременно.

11.1. Проверьте, нет ли Node

node --version
npm --versionbash

11.2. Установите nvm

curl -o- https://raw.githubusercontent.com/nvm-sh/nvm/v0.40.1/install.sh | bashbash

Если curl не установлен — попробуйте wget -qO- .... Скрипт добавит строки в ~/.bashrc. Активируйте их:

source ~/.bashrc
nvm --version   # 0.40.1bash

11.3. Установите LTS Node

nvm install --lts
nvm use --lts
nvm alias default 'lts/*'

node --version   # v20.x или v22.x
npm --version    # 10.x или вышеbash

Плюс nvm: всё лежит в ~/.nvm, никакого sudo. Хотите другую версию Node — nvm install 18 и nvm use 18.

13. Настройка прокси Vite к Django API

Чтобы React (порт 5173) мог обращаться к Django (порт 8000) без CORS-проблем, настроим прокси в Vite.

Откройте frontend/vite.config.js:

import { defineConfig } from 'vite'
import react from '@vitejs/plugin-react'

export default defineConfig({
  plugins: [react()],
  server: {
    proxy: {
      '/api': {
        target: 'http://127.0.0.1:8000',
        changeOrigin: true,
      },
    },
  },
})javascript

Теперь из React можно писать axios.get('/api/notes/') — Vite переправит на Django.

🔁 Перезапустите npm run dev после изменения vite.config.js — конфиг читается только при старте.

14.1. axios-клиент с интерцепторами src/api/client.js

import axios from 'axios';

const api = axios.create({
  baseURL: '/api',
});

// ────────────────────────────────────────────
// REQUEST-интерцептор: добавляем Bearer-токен
// ────────────────────────────────────────────
api.interceptors.request.use((config) => {
  const token = localStorage.getItem('access_token');
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});

// ────────────────────────────────────────────
// RESPONSE-интерцептор: обновляем access при 401
// ────────────────────────────────────────────
api.interceptors.response.use(
  (response) => response,
  async (error) => {
    const original = error.config;

    // Если 401 и мы ещё не пробовали обновить
    if (error.response?.status === 401 && !original._retry) {
      original._retry = true;
      const refresh = localStorage.getItem('refresh_token');

      if (refresh) {
        try {
          const { data } = await axios.post('/api/token/refresh/', { refresh });
          localStorage.setItem('access_token', data.access);
          original.headers.Authorization = `Bearer ${data.access}`;
          return api(original);   // ← повторяем исходный запрос
        } catch (e) {
          // refresh тоже протух — разлогиниваем
          localStorage.removeItem('access_token');
          localStorage.removeItem('refresh_token');
          window.location.href = '/login';
        }
      }
    }
    return Promise.reject(error);
  }
);

export default api;javascript

Как это работает: интерцепторы — это middleware для axios. Request-интерцептор выполняется до каждого запроса, response-интерцептор — после каждого ответа. Мы используем их, чтобы не дублировать логику работы с токенами в каждом компоненте.

14.2. Контекст аутентификации src/context/AuthContext.jsx

import { createContext, useContext, useState } from 'react';
import api from '../api/client';

const AuthContext = createContext(null);

export function AuthProvider({ children }) {
  const [token, setToken] = useState(
    localStorage.getItem('access_token')
  );

  const login = async (username, password) => {
    const { data } = await api.post('/token/', { username, password });
    localStorage.setItem('access_token', data.access);
    localStorage.setItem('refresh_token', data.refresh);
    setToken(data.access);
  };

  const logout = () => {
    localStorage.removeItem('access_token');
    localStorage.removeItem('refresh_token');
    setToken(null);
  };

  return (
    <AuthContext.Provider value={{ token, login, logout }}>
      {children}
    </AuthContext.Provider>
  );
}

export const useAuth = () => useContext(AuthContext);jsx

14.4. Страница входа src/pages/Login.jsx

import { useState } from 'react';
import { useNavigate } from 'react-router-dom';
import { useAuth } from '../context/AuthContext';

export default function Login() {
  const [username, setUsername] = useState('');
  const [password, setPassword] = useState('');
  const [error, setError] = useState('');
  const { login } = useAuth();
  const navigate = useNavigate();

  const handleSubmit = async (e) => {
    e.preventDefault();
    setError('');
    try {
      await login(username, password);
      navigate('/notes');
    } catch (err) {
      setError('Неверный логин или пароль');
    }
  };

  return (
    <div style={{ maxWidth: 400, margin: '50px auto', padding: 20 }}>
      <h1>Вход</h1>
      <form onSubmit={handleSubmit}>
        <input
          placeholder="Логин"
          value={username}
          onChange={(e) => setUsername(e.target.value)}
          style={{ display: 'block', width: '100%', marginBottom: 8, padding: 10 }}
        />
        <input
          type="password"
          placeholder="Пароль"
          value={password}
          onChange={(e) => setPassword(e.target.value)}
          style={{ display: 'block', width: '100%', marginBottom: 8, padding: 10 }}
        />
        <button type="submit" style={{ padding: '10px 20px' }}>
          Войти
        </button>
      </form>
      {error && <p style={{ color: 'red' }}>{error}</p>}
    </div>
  );
}jsx

14.5. Страница заметок src/pages/Notes.jsx

import { useEffect, useState } from 'react';
import api from '../api/client';
import { useAuth } from '../context/AuthContext';

export default function Notes() {
  const [notes, setNotes] = useState([]);
  const [title, setTitle] = useState('');
  const [content, setContent] = useState('');
  const { logout } = useAuth();

  const load = async () => {
    try {
      const { data } = await api.get('/notes/');
      setNotes(data);
    } catch (e) {
      console.error('Не удалось загрузить заметки', e);
    }
  };

  useEffect(() => { load(); }, []);

  const create = async (e) => {
    e.preventDefault();
    if (!title || !content) return;
    await api.post('/notes/', { title, content });
    setTitle('');
    setContent('');
    load();
  };

  const remove = async (id) => {
    if (!confirm('Удалить заметку?')) return;
    await api.delete(`/notes/${id}/`);
    load();
  };

  return (
    <div style={{ maxWidth: 700, margin: '30px auto', padding: 20 }}>
      <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center' }}>
        <h1>Мои заметки</h1>
        <button onClick={logout}>Выйти</button>
      </div>

      <form onSubmit={create} style={{ marginBottom: 20 }}>
        <input
          placeholder="Заголовок"
          value={title}
          onChange={(e) => setTitle(e.target.value)}
          style={{ display: 'block', width: '100%', marginBottom: 8, padding: 10 }}
        />
        <textarea
          placeholder="Содержание"
          value={content}
          onChange={(e) => setContent(e.target.value)}
          rows={4}
          style={{ display: 'block', width: '100%', marginBottom: 8, padding: 10 }}
        />
        <button type="submit">Добавить</button>
      </form>

      <ul style={{ listStyle: 'none', padding: 0 }}>
        {notes.map((n) => (
          <li key={n.id} style={{ border: '1px solid #ddd', padding: 16,
                                    marginBottom: 12, borderRadius: 8 }}>
            <strong>{n.title}</strong> <em>({n.owner})</em>
            <p>{n.content}</p>
            <button onClick={() => remove(n.id)}>Удалить</button>
          </li>
        ))}
      </ul>
    </div>
  );
}jsx

14.6. App.jsx — роутинг

import { BrowserRouter, Routes, Route, Navigate } from 'react-router-dom';
import { AuthProvider } from './context/AuthContext';
import PrivateRoute from './components/PrivateRoute';
import Login from './pages/Login';
import Notes from './pages/Notes';

export default function App() {
  return (

      </BrowserRouter>
    </AuthProvider>
  );
}jsx

15. Что такое JWT — разбор на примере jwt.io

Откройте jwt.io — это официальный сайт со встроенным декодером JWT.

15.1. Скопируйте свой access-токен

После логина откройте DevTools (F12) → Application → Local Storage → скопируйте значение access_token. Или через curl:

curl -X POST http://127.0.0.1:8000/api/token/ \
  -H "Content-Type: application/json" \
  -d '{"username": "ваш_логин", "password": "ваш_пароль"}'bash

15.2. Вставьте токен в jwt.io

Вы увидите три части, разделённые точками:

eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9 . eyJ0b2tlbl90eXBlIjoiYWNjZXNzIiwiZXhwIjoxNzM4... . SflKxwRJSMeKKF2QT4fwpMeJf36POk6yJV_adQssw5c
       HEADER                                        PAYLOAD                              SIGNATUREtext
  • Header — алгоритм подписи (HS256) и тип токена (JWT)
  • Payload — данные: user_id, exp, iat, token_type
  • Signature — подпись. Если поменять хоть один символ в payload — подпись станет невалидной

15.3. Что внутри

Header:

{
  "alg": "HS256",
  "typ": "JWT"
}json

Payload (у SimpleJWT по умолчанию):

{
  "token_type": "access",
  "exp": 1738617600,
  "iat": 1738614000,
  "jti": "abc123...",
  "user_id": 1
}json

Не храните в JWT секретные данные — payload легко декодируется любым человеком (сайт jwt.io это демонстрирует). Подпись лишь гарантирует, что токен выдан вашим сервером и не был изменён.

17. Запуск одним скриптом

Создайте файл ~/projects/fullstack-app/run-all.sh:

#!/bin/bash
# Запуск Django и React одновременно в текущем терминале

set -e
cd "$(dirname "$0")"

echo "🚀 Запускаем Django-бэкенд..."
cd backend
source ../venv/bin/activate
python manage.py runserver &
DJANGO_PID=$!
cd ..

echo "⚛️  Запускаем React-фронтенд..."
cd frontend
npm run dev &
REACT_PID=$!
cd ..

echo ""
echo "✅ Оба сервера запущены!"
echo "   Backend:  http://127.0.0.1:8000"
echo "   Frontend: http://localhost:5173"
echo ""
echo "Нажмите Ctrl+C для остановки обоих."

# При Ctrl+C убиваем оба процесса
trap "kill $DJANGO_PID $REACT_PID 2>/dev/null; exit" INT TERM
waitbash

Сделайте его исполняемым и запустите:

chmod +x run-all.sh
./run-all.shbash
